예외 처리

//자바에서는 예외처리를 try, catch로 한다
	try {
		int[] arr = new int[5];
			
		for(int i = 0; i < 10; i++) {
			arr[i] = 10 * (i + 1);
			System.out.println(arr[i]);
		}
		
	} catch (ArrayIndexOutOfBoundsException e) {
		System.out.println("배열의 인덱스 범위 초과!");
		// 여러개의 예외 상황을 추가 할 수 있다.
	} catch (IllegalFormatConversionException e) {
		System.out.println("숫자양식 예외 발생!");
	} catch (Exception e) {
		System.out.println("포괄적인 예외 발생!");
	} finally {
		System.out.println("??????");
	}

//	전역변수
//	  전역 -> 전 지역 
//	전 지역에서 사용 가능한 변수 
//	변수의 생명주기가 전 지역에서 사용 가능하게끔 설정되어있다.
//	자바는 실질적인 전역변수가 존재하지 않는다.
//	자바는 클래스 없이는 코드를 작성할 수 없다.
//	자바에서 전역변수의 개념을 대체할 수 있는 변수는 클래스변수(static변수)라고 한다.

//	멤버변수, 필드 
//	클래스 안에서, 메서드 밖에서 선언된 변수
//	같은 클래스 내부의 전 지역에서 사용 가능
//	변수를 선언하고 초기화를 하지 않으면 기본값으로 초기화 된다.
		
		
//	지역변수
//	지역 --> 제한된 지역 
//	메서드 안에서 선언된 변수, 코드 블럭(클래스 코드블럭 제외) 내에서 선언된 변수 
//	제한된 지역(본인이 만들어진 지역)에서 사용 가능한 변수 
//	지역변수는 값을 초기화 하지 않으면 사용할 수 없다.
	
//	같은 공간 내에서 동일한 이름으로 변수를 선언할 수 없다.
