class Main {
  public static void main(String[] args) {
    자동차 a자동차 = new 자동차();

    a자동차.자동차2 = "아반테";
    a자동차.자동차3 = "소나타";
    a자동차.속도1 = 200;
    a자동차.속도2 = 160;

    a자동차.번호3 = 30;
	  
    for(int i = 1; i <= 3; i++) {
    	System.out.println(a자동차.자동차1 + "가 달립니다.");
    }
    System.out.println();
    System.out.println(a자동차.자동차1 + "가 달립니다.");
    System.out.println(a자동차.자동차2 + "가 달립니다.");
    System.out.println(a자동차.자동차3 + "가 달립니다.");
    System.out.println();
    
    System.out.println(a자동차.자동차1 + "가 최고속력 " + a자동차.속도1 +"로 달립니다.");
    System.out.println(a자동차.자동차2 + "가 최고속력 " + a자동차.속도2 +"로 달립니다.");
    System.out.println(a자동차.자동차3 + "가 최고속력 " + a자동차.속도3 +"로 달립니다.");
    System.out.println();
    
    System.out.println(a자동차.번호1 + "번 " + a자동차.자동차1 + "가 최고속력 " + a자동차.속도1 +"로 달립니다.");
    System.out.println(a자동차.번호2 + "번 " + a자동차.자동차2 + "가 최고속력 " + a자동차.속도2 +"로 달립니다.");
    System.out.println(a자동차.번호3 + "번 " + a자동차.자동차3 + "가 최고속력 " + a자동차.속도3 +"로 달립니다.");
    
  }
}
