		Scanner sc = new Scanner(System.in);
		String cmd = sc.nextLine();
		
		String[] str = cmd.split(" ");
		int a = Integer.parseInt(str[0]); 
		int b = Integer.parseInt(str[1]);
		
		for(int i = 1; i <= a; i++) {
			for(int j = 1; j <= b; j++) {
				System.out.printf("*");
			}
			System.out.println();
		}
		
		
	}
}

package java_exam;

import java.util.Scanner;

public class Main {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		int num = sc.nextInt();
		for(int i = num; i >= 1; i--) {
			System.out.printf(" ".repeat(i -1));
			System.out.println("*".repeat(2 * (num - i) + 1));
		}
		for(int i = 1; i <= num; i++) {
			System.out.printf(" ".repeat(i -1));
			System.out.println("*".repeat(2 * (num - i) + 1));
		}
		System.out.println();
		
		for(int i = 1; i <= num; i++) {
			for(int j = 1; j <= i - 1; j++) {
				System.out.printf(" ");
			}
			for(int j = 1; j <= 2 * (num - i) + 1; j++) {
				System.out.printf("*");
			}
			System.out.println();
		}
		System.out.println("===============");
		for(int i = num; i >= 1; i--) {
			for(int j = i - 1; j >= 1; j--) {
				System.out.printf(" ");
			}
			for(int j = 2 * (num - i) + 1 ; j >= 1; j--) {
				System.out.printf("*");
			}
			System.out.println();
		}
		sc.close();
	}
}

package com.koreaIT.exam;


import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;


public class Main {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		System.out.println("== 프로그램 시작 ==");
		int LastArticleId = 0;
		List<Article> articles = new ArrayList<>();
		
		while (true) {
			System.out.printf("명령어 : ");
			String cmd = sc.nextLine();
			if (cmd.equals("exit")) {
				break;
			}
			if(cmd.equals("article list")) {
				if(articles.size() == 0) {
					System.out.println("게시글이 존재하지 않습니다.");
					continue;
				}
				System.out.println("번호	|	제목");
				for(int i = articles.size() - 1; i >= 0; i--) {
					Article article = articles.get(i);
					System.out.printf("%s	|	%s\n", article.id, article.title);
				}
				
				
			} else if(cmd.equals("article write") ) {
				System.out.printf("제목 : ");
				String title = sc.nextLine();
				System.out.printf("내용 : ");
				String body = sc.nextLine();
				
				LastArticleId++;
				
				Article article = new Article(body,LastArticleId,title);
				articles.add(article);
				
				System.out.println(LastArticleId + "번 게시물이 작성되었습니다.");
			}
		}
		System.out.println("== 프로그램 종료 ==");
		
		sc.close();
	}
}
class Article {
	Article(String body, int id, String title) {
		this.body = body;
		this.id = id;
		this.title = title;
	}
	int id;
	String title;
	String body;
}
