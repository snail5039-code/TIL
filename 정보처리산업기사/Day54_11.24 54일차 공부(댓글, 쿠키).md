## 💬 댓글 CRUD 비동기 처리 코드 예시
### 1. JavaScript / jQuery (AJAX 요청 및 DOM 조작)
### 1-1. 댓글 목록 갱신 및 동적 이벤트 바인딩
\$\\text\{jQuery\}\$의 \$\\text\{.on()\}\$을 사용하여 동적으로 생성된 **수정/삭제 버튼**에 이벤트를 바인딩하고, 템플릿 리터럴을 사용해 \$\\text\{HTML\}\$을 생성합니다.
JavaScript
```java
// 댓글 목록을 서버로부터 받아와 HTML을 구성하고 화면에 표시하는 함수
function displayComments(data) {
    const container = $('#commentListContainer');
    // 컨테이너 초기화
    container.empty(); 
    
    // JS map()과 join('')을 사용하여 문자열 누적 대신 효율적인 HTML 생성
    const html = data.map(comment => {
        // 백틱(템플릿 리터럴)을 사용한 가독성 높은 HTML 생성
        return `
            <div class="comment-item" id="comment-${comment.id}">
                <p class="comment-content" data-id="${comment.id}">${comment.content}</p>
                <p class="comment-author">${comment.author}</p>
                
                <button class="btn-edit" data-id="${comment.id}" data-content="${comment.content}">수정</button>
                <button class="btn-delete" data-id="${comment.id}">삭제</button>
                
                <div class="edit-form" data-id="${comment.id}" style="display:none;">
                    <textarea class="edit-textarea">${comment.content}</textarea>
                    <button class="btn-save" data-id="${comment.id}">저장</button>
                    <button class="btn-cancel">취소</button>
                </div>
            </div>
        `;
    }).join(''); // 배열의 모든 요소를 하나의 문자열로 합침

    container.html(html);
}

// 📌 동적으로 생성된 버튼에 이벤트 바인딩 (하나의 컨테이너에 위임)
$(document).ready(function() {
    const container = $('#commentListContainer');

    // 1. 댓글 삭제 이벤트 🗑️
    container.on('click', '.btn-delete', function() {
        const commentId = $(this).data('id');
        if (confirm('정말 삭제하시겠습니까?')) {
            deleteComment(commentId);
        }
    });

    // 2. 댓글 수정 폼 열기 이벤트 ✏️
    container.on('click', '.btn-edit', function() {
        const commentId = $(this).data('id');
        $(`.edit-form[data-id=${commentId}]`).show();
        $(`.comment-content[data-id=${commentId}]`).hide();
        $(this).hide(); // 수정 버튼 숨기기
    });

    // 3. 댓글 수정 저장 이벤트 💾
    container.on('click', '.btn-save', function() {
        const commentId = $(this).data('id');
        const newContent = $(this).siblings('.edit-textarea').val();
        updateComment(commentId, newContent);
    });
    
    // 4. 댓글 수정 취소 이벤트
    container.on('click', '.btn-cancel', function() {
        const commentId = $(this).closest('.edit-form').data('id');
        $(`.edit-form[data-id=${commentId}]`).hide();
        $(`.comment-content[data-id=${commentId}]`).show();
        $(`.btn-edit[data-id=${commentId}]`).show();
    });
});
```

```java
// 1. 댓글 작성 AJAX 📝
function saveComment() {
    const content = $('#newCommentContent').val();
    const author = $('#newCommentAuthor').val(); 
    
    $.ajax({
        url: '/api/comments',
        type: 'POST', // INSERT (작성)
        contentType: 'application/json',
        data: JSON.stringify({ content: content, author: author }),
        success: function(response) {
            if (response.success) {
                alert('댓글이 작성되었습니다.');
                // 폼 초기화
                $('#newCommentContent').val(''); 
                // 목록 다시 로드 (혹은 새로 작성된 댓글만 DOM에 추가)
                loadComments(); 
            }
        }
    });
}

// 2. 댓글 삭제 AJAX 🗑️
function deleteComment(commentId) {
    $.ajax({
        url: `/api/comments/${commentId}`, 
        type: 'DELETE', // DELETE
        success: function(response) {
            if (response.success) {
                // DOM에서 해당 댓글 요소 즉시 제거
                $(`#comment-${commentId}`).remove(); 
            } else {
                alert('삭제 실패: ' + response.message);
            }
        }
    });
}

// 3. 댓글 수정 AJAX ✏️
function updateComment(commentId, newContent) {
    $.ajax({
        url: `/api/comments/${commentId}`,
        type: 'PUT', // UPDATE (수정)
        contentType: 'application/json',
        data: JSON.stringify({ id: commentId, content: newContent }),
        success: function(response) {
            if (response.success) {
                alert('댓글이 수정되었습니다.');
                // DOM 갱신: 수정된 내용을 즉시 반영하고 폼 숨기기
                $(`.comment-content[data-id=${commentId}]`).text(newContent).show();
                $(`.edit-form[data-id=${commentId}]`).hide();
                $(`.btn-edit[data-id=${commentId}]`).show();
            } else {
                alert('수정 실패: ' + response.message);
            }
        }
    });
}
```

### 2-2. MyBatis Mapper XML (DAO에 직접 쿼리)
XML
```java
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.example.dao.CommentDAO">
    
    <insert id="insertComment" parameterType="com.example.domain.Comment" useGeneratedKeys="true" keyProperty="id">
        INSERT INTO comments (content, author, regdate)
        VALUES (#{content}, #{author}, NOW())
    </insert>
    
    <delete id="deleteComment" parameterType="int">
        DELETE FROM comments
        WHERE id = #{commentId}
    </delete>
    
    <update id="updateComment" parameterType="com.example.domain.Comment">
        UPDATE comments
        SET 
            content = #{content},
            updatedate = NOW()
        WHERE id = #{id}
    </update>
    
</mapper>
```
