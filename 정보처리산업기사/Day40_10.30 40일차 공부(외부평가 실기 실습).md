package HRD_1234;

import java.io.IOException;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.Statement;
import java.util.ArrayList;
import java.util.List;

import HRD_1234.dto.Car;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

@WebServlet("/usr/car/carDetail")
public class CarDetail extends HttpServlet {
	private static final long serialVersionUID = 1L;

	protected void doGet(HttpServletRequest request, HttpServletResponse response)
			throws ServletException, IOException {
		try {
			Class.forName("oracle.jdbc.OracleDriver");
			Connection con = DriverManager.getConnection("jdbc:oracle:thin:@//localhost:1521/xe", "system", "1234");
			Statement stmt = con.createStatement();
			ResultSet rs = stmt.executeQuery("select * from TBL_CAR");

			List<Car> cars = new ArrayList<>();

			while (rs.next()) {
				String carNo = rs.getString("CAR_NO");
				String carModel = rs.getString("CAR_MODEL");
				String carCompany = rs.getString("CAR_COMPANY");
				String carDate = rs.getString("CAR_DATE");

				cars.add(new Car(carNo, carModel, carCompany, carDate));
			}
			stmt.close();
			con.close();

			request.setAttribute("cars", cars);

		} catch (Exception e) {
			e.printStackTrace();
		}

		request.getRequestDispatcher("/cardetail.jsp").forward(request, response);
	}

}

package HRD_1234;

import java.io.IOException;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.Statement;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

@WebServlet("/usr/car/doCarDrive")
public class DoCarDrive extends HttpServlet {
	private static final long serialVersionUID = 1L;

	protected void doGet(HttpServletRequest request, HttpServletResponse response)
			throws ServletException, IOException {
		try {
			Class.forName("oracle.jdbc.OracleDriver");
			Connection con = DriverManager.getConnection("jdbc:oracle:thin:@//localhost:1521/xe", "system", "1234");
			Statement stmt = con.createStatement();

			String drvDate = request.getParameter("DRV_DATE");
			String carNo = request.getParameter("CAR_NO");
			String drvStart = request.getParameter("DRV_START");
			String drvEnd = request.getParameter("DRV_END");
			String deptCode = request.getParameter("DEPT_CODE");
			String drvMoney = request.getParameter("DRV_MONEY");
			
			

			stmt.executeUpdate(
					"insert into TBL_DRIVE (DRV_DATE, CAR_NO, DRV_START, DRV_END, DEPT_CODE, DRV_MONEY) VALUES('"
							+ drvDate + "', '" + carNo + "', " + drvStart + ", " + drvEnd + ", '" + deptCode + "', "+ drvMoney + ")");

			stmt.close();
			con.close();

		} catch (Exception e) {
			e.printStackTrace();
		}
		
		response.setContentType("text/html charset=UTF-8");
		response.getWriter().append("<script>alert('주행내역이 정상적으로 등록 되었습니다!'); location.replace('/usr/car/main');</script>");
	}

}

package HRD_1234.dto;

public class Car {
	private String CAR_NO;
	private String CAR_MODEL;
	private String CAR_COMPANY;
	private String CAR_DATE;

	public Car(String CAR_NO, String CAR_MODEL, String CAR_COMPANY, String CAR_DATE) {
		this.CAR_NO = CAR_NO;
		this.CAR_MODEL = CAR_MODEL;
		this.CAR_COMPANY = CAR_COMPANY;
		this.CAR_DATE = CAR_DATE;
	}

	public String getCAR_NO() {
		return CAR_NO;
	}

	public void setCAR_NO(String cAR_NO) {
		CAR_NO = cAR_NO;
	}

	public String getCAR_MODEL() {
		return CAR_MODEL;
	}

	public void setCAR_MODEL(String cAR_MODEL) {
		CAR_MODEL = cAR_MODEL;
	}

	public String getCAR_COMPANY() {
		return CAR_COMPANY;
	}

	public void setCAR_COMPANY(String cAR_COMPANY) {
		CAR_COMPANY = cAR_COMPANY;
	}

	public String getCAR_DATE() {
		return CAR_DATE;
	}

	public void setCAR_DATE(String cAR_DATE) {
		CAR_DATE = cAR_DATE;
	}

}
